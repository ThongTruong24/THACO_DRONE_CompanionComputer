cmake_minimum_required(VERSION 3.16)
project(cc_telemetry)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(mavros_msgs REQUIRED)
find_package(cc_interfaces REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(nlohmann_json REQUIRED)

add_library(cc_core src/parameter_store.cpp src/collectors.cpp src/system_metrics.cpp)
target_include_directories(cc_core PUBLIC include)
target_link_libraries(cc_core yaml-cpp nlohmann_json::nlohmann_json)

add_executable(cc_telemetry_node src/node.cpp)
target_link_libraries(cc_telemetry_node cc_core)
ament_target_dependencies(cc_telemetry_node rclcpp std_msgs mavros_msgs cc_interfaces)

install(TARGETS cc_core ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(TARGETS cc_telemetry_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/)

if(BUILD_TESTING)
 find_package(ament_cmake_gtest REQUIRED)
 ament_add_gtest(core_test test/core_test.cpp)
 target_link_libraries(core_test cc_core)
endif()
ament_package()
