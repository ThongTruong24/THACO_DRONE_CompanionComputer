cmake_minimum_required(VERSION 3.20)
project(drone_vision_new LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(THACO_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../cc-agent/mavlink" CACHE PATH "Existing generated THACO headers")
set(JSON_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../cc-agent/include" CACHE PATH "Existing nlohmann JSON header")
find_package(Threads REQUIRED)
add_library(vision_transport bridge.cpp camera_protocol.cpp)
target_include_directories(vision_transport PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_include_directories(vision_transport SYSTEM PUBLIC ${THACO_INCLUDE_DIR})
target_link_libraries(vision_transport PUBLIC Threads::Threads)
target_compile_options(vision_transport PRIVATE -Wall -Wextra -Wpedantic)
add_executable(mavlink-bridge mavlink_bridge.cpp)
target_include_directories(mavlink-bridge SYSTEM PRIVATE ${JSON_INCLUDE_DIR})
target_link_libraries(mavlink-bridge PRIVATE vision_transport)
install(TARGETS mavlink-bridge DESTINATION bin)
option(BUILD_TESTING "Build isolated Vision tests" ON)
if(BUILD_TESTING)
  enable_testing()
  find_package(GTest REQUIRED)
  include(GoogleTest)
  foreach(category IN ITEMS mavlink_codec camera_protocol tcp_transport)
    add_executable(test_${category} ../tests/test_${category}.cpp)
    target_link_libraries(test_${category} PRIVATE vision_transport GTest::gtest_main)
    if(NOT category STREQUAL "tcp_transport")
      gtest_discover_tests(test_${category} PROPERTIES LABELS unit)
    else()
      gtest_discover_tests(test_${category} PROPERTIES LABELS tcp)
    endif()
  endforeach()
  add_executable(router-probe ../tests/router_probe.cpp)
  target_link_libraries(router-probe PRIVATE vision_transport)
  add_executable(packet-fixture ../tests/packet_fixture.cpp)
  target_include_directories(packet-fixture SYSTEM PRIVATE ${JSON_INCLUDE_DIR})
  target_link_libraries(packet-fixture PRIVATE vision_transport)
  add_executable(render-baseline-config ../tests/render_baseline_config.cpp
    ../../mavlink-router-controller/src/router_supervisor.cpp)
  target_include_directories(render-baseline-config PRIVATE ../../mavlink-router-controller/include)
  target_link_libraries(render-baseline-config PRIVATE Threads::Threads)
  set(QGC_DIALECT_INCLUDE_DIR "" CACHE PATH "Generated QGC all dialect directory; no Pi-header fallback")
  set(QGC_DIALECT_REVISION "" CACHE STRING "QGC MAVLink dependency commit used by the fixture")
  if(QGC_DIALECT_INCLUDE_DIR)
    if(NOT EXISTS "${QGC_DIALECT_INCLUDE_DIR}/mavlink.h" OR NOT QGC_DIALECT_REVISION)
      message(FATAL_ERROR "QGC fixture requires all/mavlink.h and its explicit dependency revision")
    endif()
    add_executable(qgc-camera-packets ../tests/qgc_camera_packets.cpp)
    target_include_directories(qgc-camera-packets SYSTEM PRIVATE ${QGC_DIALECT_INCLUDE_DIR} ${JSON_INCLUDE_DIR})
    target_compile_definitions(qgc-camera-packets PRIVATE QGC_DIALECT_REVISION="${QGC_DIALECT_REVISION}")
  endif()
endif()
